example {V : Type*} [NormedAddCommGroup V] [NormedSpace ℝ V] [FiniteDimensional ℝ V]
    {n : ℕ} (L : V ≃L[ℝ] (Fin n → ℝ)) :
    DiskCylinder.Disk (E := V) ≃ₜ (Fin n → unitInterval) :=
  C15_DiskCubeInterfaceCheck.homeomorph L

example {V : Type*} [NormedAddCommGroup V] [NormedSpace ℝ V] [FiniteDimensional ℝ V]
    {n : ℕ} (L : V ≃L[ℝ] (Fin n → ℝ)) (z : DiskCylinder.Disk (E := V)) :
    C15_DiskCubeInterfaceCheck.homeomorph L z ∈ Cube.boundary (Fin n) ↔
      ‖(z : V)‖ = 1 :=
  C15_DiskCubeInterfaceCheck.boundary_iff L z
